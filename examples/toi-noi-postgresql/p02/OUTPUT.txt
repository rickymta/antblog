-- [lab] ===== p01/01-schema.sql =====

-- P01 · 01-schema.sql — lược đồ tối thiểu cho thành phố, khách, tài xế, quán,
-- chuyến và lịch sử trạng thái chuyến (PG-003, PG-004, PG-005).
--
-- Bám lược đồ lõi trong hồ sơ `du-an-mau-toi-noi`: cùng tên bảng, tên cột,
-- kiểu dữ liệu và giá trị mặc định. Khác biệt có chủ đích (xem p01/README.md):
--   1. Mọi ràng buộc đều được ĐẶT TÊN (pk_…, fk_…, uq_…, ck_…) để thông báo lỗi
--      chỉ thẳng vào quy tắc bị vi phạm.
--   2. Thêm các CHECK mã hoá quy tắc nghiệp vụ (định dạng số điện thoại, cước
--      không âm, thứ tự mốc thời gian, trạng thái ↔ tài xế ↔ mốc hoàn thành…).
--   3. `trip_events.status` có CHECK cùng danh sách 8 trạng thái với `trips`.
--   4. Chưa có `driver_locations` (thuộc các phần sau, không cần cho P01).
--
-- Quy ước kiểu dữ liệu (PG-004):
--   * Tiền: số nguyên ĐỒNG (`integer`). VND không dùng đơn vị lẻ, nên không cần
--     numeric/float. `integer` chứa tới 2.147.483.647 đ cho MỘT chuyến — dư sức;
--     `sum(integer)` trả về `bigint`, nên cộng hàng triệu chuyến vẫn không tràn.
--     Không dùng `real`/`double precision` (sai số nhị phân) hay kiểu `money`
--     (định dạng phụ thuộc lc_monetary của server).
--   * Thời điểm: `timestamptz` — lưu một khoảnh khắc tuyệt đối, hiển thị theo
--     TimeZone của phiên. Không dùng `timestamp` (không múi giờ) cho mốc sự kiện.
--   * Định danh: `uuid` sinh bằng gen_random_uuid() (UUID v4, có sẵn từ PG 13,
--     không cần extension) + mã số dễ đọc (`code`) cho người vận hành.

SET search_path TO toinoi;
SET

SET TIME ZONE 'Asia/Ho_Chi_Minh';

-- ---------------------------------------------------------------------------
-- Thành phố: danh mục nhỏ, id do người quản trị đặt (1 = SG, 2 = HN…).
CREATE TABLE cities (
  id    smallint NOT NULL,
  code  text     NOT NULL,
  name  text     NOT NULL,
  CONSTRAINT pk_cities PRIMARY KEY (id),
  CONSTRAINT uq_cities_code UNIQUE (code),
  CONSTRAINT ck_cities_code_format CHECK (code ~ '^[A-Z]{2}$'),
  CONSTRAINT ck_cities_name_not_blank CHECK (btrim(name) <> '')
);
CREATE TABLE

-- ---------------------------------------------------------------------------
-- Khách hàng: đăng nhập bằng số điện thoại ⇒ số điện thoại là duy nhất.
CREATE TABLE customers (
  id         uuid        NOT NULL DEFAULT gen_random_uuid(),
  phone      text        NOT NULL,
  full_name  text        NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT pk_customers PRIMARY KEY (id),
  CONSTRAINT uq_customers_phone UNIQUE (phone),
  -- Lưu MỘT dạng chuẩn: 10 chữ số bắt đầu bằng 0. Nếu cho phép cả '+84…' và
  -- '0…', UNIQUE không còn chặn được cùng một số viết theo hai cách.
  CONSTRAINT ck_customers_phone_format CHECK (phone ~ '^0[0-9]{9}$'),
  CONSTRAINT ck_customers_full_name_not_blank CHECK (btrim(full_name) <> '')
);
CREATE TABLE

-- ---------------------------------------------------------------------------
-- Tài xế: `code` là mã nội bộ do bộ phận tuyển dụng cấp (8001, 8002…).
CREATE TABLE drivers (
  id           uuid        NOT NULL DEFAULT gen_random_uuid(),
  code         integer     NOT NULL,
  full_name    text        NOT NULL,
  city_id      smallint    NOT NULL,
  vehicle_type text        NOT NULL,
  status       text        NOT NULL DEFAULT 'offline',
  created_at   timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT pk_drivers PRIMARY KEY (id),
  CONSTRAINT uq_drivers_code UNIQUE (code),
  CONSTRAINT fk_drivers_city FOREIGN KEY (city_id) REFERENCES cities (id),
  CONSTRAINT ck_drivers_code_positive CHECK (code > 0),
  CONSTRAINT ck_drivers_full_name_not_blank CHECK (btrim(full_name) <> ''),
  CONSTRAINT ck_drivers_vehicle_type CHECK (vehicle_type IN ('bike', 'car')),
  CONSTRAINT ck_drivers_status CHECK (status IN ('offline', 'online', 'busy'))
);
CREATE TABLE

-- ---------------------------------------------------------------------------
-- Quán: cần cho quy tắc "đơn đồ ăn phải có quán". P01 chỉ giữ các cột lõi.
CREATE TABLE merchants (
  id         uuid        NOT NULL DEFAULT gen_random_uuid(),
  city_id    smallint    NOT NULL,
  name       text        NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT pk_merchants PRIMARY KEY (id),
  CONSTRAINT fk_merchants_city FOREIGN KEY (city_id) REFERENCES cities (id),
  CONSTRAINT ck_merchants_name_not_blank CHECK (btrim(name) <> '')
);
CREATE TABLE

-- ---------------------------------------------------------------------------
-- Chuyến: một yêu cầu đặt xe / giao đồ ăn / giao bưu kiện.
--
-- Lưu ý khi đọc các CHECK bên dưới: CHECK chỉ TỪ CHỐI khi biểu thức ra FALSE.
-- Nếu biểu thức ra NULL (vd. so sánh với accepted_at đang NULL) thì dòng được
-- CHẤP NHẬN. Vì vậy `accepted_at >= requested_at` tự bỏ qua chuyến chưa có
-- người nhận, còn các quy tắc "phải có" được viết tường minh bằng IS NULL /
-- IS NOT NULL.
CREATE TABLE trips (
  id           uuid        NOT NULL DEFAULT gen_random_uuid(),
  code         bigint      GENERATED ALWAYS AS IDENTITY (START WITH 100000),
  kind         text        NOT NULL,
  city_id      smallint    NOT NULL,
  customer_id  uuid        NOT NULL,
  driver_id    uuid,
  merchant_id  uuid,
  status       text        NOT NULL DEFAULT 'requested',
  fare         integer     NOT NULL,              -- đồng
  requested_at timestamptz NOT NULL DEFAULT now(),
  accepted_at  timestamptz,
  completed_at timestamptz,
  updated_at   timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT pk_trips PRIMARY KEY (id),
  CONSTRAINT uq_trips_code UNIQUE (code),
  CONSTRAINT fk_trips_city     FOREIGN KEY (city_id)     REFERENCES cities (id),
  CONSTRAINT fk_trips_customer FOREIGN KEY (customer_id) REFERENCES customers (id),
  CONSTRAINT fk_trips_driver   FOREIGN KEY (driver_id)   REFERENCES drivers (id),
  CONSTRAINT fk_trips_merchant FOREIGN KEY (merchant_id) REFERENCES merchants (id),
  CONSTRAINT ck_trips_kind CHECK (kind IN ('ride', 'food', 'parcel')),
  CONSTRAINT ck_trips_status CHECK (status IN
    ('requested', 'searching', 'accepted', 'arriving', 'picked_up', 'in_transit', 'completed', 'cancelled')),
  -- 0 đ hợp lệ (chuyến miễn phí do khuyến mãi); số âm thì không.
  CONSTRAINT ck_trips_fare_non_negative CHECK (fare >= 0),
  -- Đơn đồ ăn phải có quán; đặt xe và bưu kiện không gắn quán.
  CONSTRAINT ck_trips_merchant_only_food CHECK ((kind = 'food') = (merchant_id IS NOT NULL)),
  -- Có tài xế ⇔ có mốc nhận chuyến (kể cả chuyến đã nhận rồi bị huỷ).
  CONSTRAINT ck_trips_driver_accepted_together CHECK ((driver_id IS NULL) = (accepted_at IS NULL)),
  -- Hai quy tắc dạng "nếu A thì B", viết là `NOT A OR B`:
  --   chưa ai nhận (requested/searching) ⇒ chưa có tài xế;
  --   từ accepted tới completed ⇒ phải có tài xế.
  -- Chuyến huỷ có thể có hoặc không có tài xế (huỷ trước hay sau khi được nhận).
  -- Mỗi CHECK chỉ giữ MỘT quy tắc: trạng thái lạ như 'done' chỉ vi phạm
  -- ck_trips_status, thông báo lỗi không bị lẫn.
  CONSTRAINT ck_trips_unassigned_no_driver CHECK (
    status NOT IN ('requested', 'searching') OR driver_id IS NULL),
  CONSTRAINT ck_trips_assigned_has_driver CHECK (
    status NOT IN ('accepted', 'arriving', 'picked_up', 'in_transit', 'completed') OR driver_id IS NOT NULL),
  -- Có mốc hoàn thành ⇔ trạng thái hoàn thành.
  CONSTRAINT ck_trips_completed_at_by_status CHECK ((status = 'completed') = (completed_at IS NOT NULL)),
  CONSTRAINT ck_trips_accepted_after_requested CHECK (accepted_at >= requested_at),
  CONSTRAINT ck_trips_completed_after_accepted CHECK (completed_at >= accepted_at)
);
CREATE TABLE

-- Ba index giống hồ sơ chung. P01 chưa cần chúng; P03 sẽ đo lợi ích và cái giá.
-- (PRIMARY KEY và UNIQUE tự tạo index riêng: pk_trips, uq_trips_code.)
CREATE INDEX ix_trips_city_requested     ON trips (city_id, requested_at);
CREATE INDEX

CREATE INDEX ix_trips_customer_requested ON trips (customer_id, requested_at DESC);
CREATE INDEX

CREATE INDEX ix_trips_driver_id          ON trips (driver_id);
CREATE INDEX

-- ---------------------------------------------------------------------------
-- Lịch sử trạng thái: mỗi lần chuyến đổi trạng thái thêm một dòng, không sửa.
CREATE TABLE trip_events (
  id          bigint      GENERATED ALWAYS AS IDENTITY,
  trip_id     uuid        NOT NULL,
  status      text        NOT NULL,
  occurred_at timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT pk_trip_events PRIMARY KEY (id),
  CONSTRAINT fk_trip_events_trip FOREIGN KEY (trip_id) REFERENCES trips (id),
  CONSTRAINT ck_trip_events_status CHECK (status IN
    ('requested', 'searching', 'accepted', 'arriving', 'picked_up', 'in_transit', 'completed', 'cancelled'))
);
CREATE TABLE

CREATE INDEX ix_trip_events_trip ON trip_events (trip_id, occurred_at);
CREATE INDEX

-- ---------------------------------------------------------------------------
-- Cây khu vực điều phối của Tới Nơi (hư cấu, KHÔNG phải đơn vị hành chính):
-- thành phố (gốc, parent_id NULL) → vùng → khu. Chuyến được gắn vào KHU.
-- id đặt tay theo quy ước dễ đọc: thành phố 1..3, vùng 11..34, khu 111..322.
CREATE TABLE areas (
  id        integer  NOT NULL,
  parent_id integer,
  code      text     NOT NULL,
  name      text     NOT NULL,
  level     text     NOT NULL,
  city_id   smallint,
  CONSTRAINT pk_areas PRIMARY KEY (id),
  CONSTRAINT uq_areas_code UNIQUE (code),
  CONSTRAINT fk_areas_parent FOREIGN KEY (parent_id) REFERENCES areas (id),
  CONSTRAINT fk_areas_city FOREIGN KEY (city_id) REFERENCES cities (id),
  CONSTRAINT ck_areas_level CHECK (level IN ('city', 'district', 'zone')),
  -- Chỉ nút gốc (thành phố) không có cha, và chỉ nút gốc gắn với cities.
  CONSTRAINT ck_areas_root_no_parent CHECK ((level = 'city') = (parent_id IS NULL)),
  CONSTRAINT ck_areas_root_has_city CHECK ((level = 'city') = (city_id IS NOT NULL)),
  -- Chặn được vòng 1 nút (tự làm cha mình)… nhưng KHÔNG chặn được vòng 2+ nút:
  -- CHECK chỉ nhìn một dòng. 19-chu-trinh.sql cho thấy hệ quả.
  CONSTRAINT ck_areas_not_own_parent CHECK (parent_id <> id)
);
CREATE TABLE

CREATE INDEX ix_areas_parent ON areas (parent_id);
CREATE INDEX

COMMENT ON TABLE areas IS 'Cây khu vực điều phối: city → district (vùng) → zone (khu); dữ liệu hư cấu';
COMMENT

-- ---------------------------------------------------------------------------
-- Đánh giá sau chuyến: tối đa MỘT dòng mỗi chuyến (khoá chính là trip_id).
--   * không có dòng          = khách chưa mở màn đánh giá
--   * stars NULL             = khách mở màn đánh giá nhưng bỏ trống phần sao
--   * tip NULL / tip = 0     = bỏ qua bước tip  /  chủ động chọn "không tip"
-- Chuyến huỷ không được đánh giá (không áp dụng) — quy tắc nhiều bảng, CHECK
-- một dòng không diễn đạt được; 02-kiem-tra-fixture.sql kiểm.
CREATE TABLE trip_ratings (
  trip_id  uuid        NOT NULL,
  stars    smallint,
  tip      integer,                       -- đồng
  rated_at timestamptz NOT NULL,
  CONSTRAINT pk_trip_ratings PRIMARY KEY (trip_id),
  CONSTRAINT fk_trip_ratings_trip FOREIGN KEY (trip_id) REFERENCES trips (id),
  CONSTRAINT ck_trip_ratings_stars CHECK (stars BETWEEN 1 AND 5),
  CONSTRAINT ck_trip_ratings_tip_non_negative CHECK (tip >= 0)
);
CREATE TABLE

COMMENT ON TABLE trip_ratings IS 'Đánh giá sau chuyến; 0..1 dòng mỗi chuyến';
COMMENT

COMMENT ON COLUMN trip_ratings.stars IS '1..5 sao; NULL = khách bỏ trống phần sao';
COMMENT

COMMENT ON COLUMN trip_ratings.tip IS 'Tiền tip (đồng); NULL = bỏ qua bước tip, 0 = chọn không tip';
COMMENT

-- ---------------------------------------------------------------------------
-- Giao dịch thanh toán: 0..n dòng mỗi chuyến.
--   kind   charge (thu) · refund (hoàn) · adjustment (điều chỉnh của CSKH, CÓ DẤU)
--   method cash (tiền mặt, tài xế ghi nhận) · wallet (ví VíXanh) · card (thẻ)
--   status pending · succeeded · failed — chỉ succeeded làm đổi tiền thật
-- Số tiền luôn là số nguyên đồng. charge/refund ghi số DƯƠNG, chiều tiền do
-- kind quyết định; adjustment ghi số có dấu (+ thu thêm, − giảm trừ).
-- Tiền thực thu của một chuyến = Σ charge − Σ refund + Σ adjustment (succeeded).
CREATE TABLE payments (
  id           bigint      GENERATED ALWAYS AS IDENTITY,
  trip_id      uuid        NOT NULL,
  kind         text        NOT NULL,
  method       text        NOT NULL,
  status       text        NOT NULL,
  amount       integer     NOT NULL,       -- đồng
  provider_ref text,                       -- mã giao dịch phía ví/cổng thẻ
  paid_at      timestamptz NOT NULL,       -- thời điểm phát sinh giao dịch
  CONSTRAINT pk_payments PRIMARY KEY (id),
  CONSTRAINT fk_payments_trip FOREIGN KEY (trip_id) REFERENCES trips (id),
  -- Cổng thanh toán không cấp trùng mã; NULL (tiền mặt) không bị UNIQUE chặn.
  CONSTRAINT uq_payments_provider_ref UNIQUE (provider_ref),
  CONSTRAINT ck_payments_kind CHECK (kind IN ('charge', 'refund', 'adjustment')),
  CONSTRAINT ck_payments_method CHECK (method IN ('cash', 'wallet', 'card')),
  CONSTRAINT ck_payments_status CHECK (status IN ('pending', 'succeeded', 'failed')),
  CONSTRAINT ck_payments_amount_sign CHECK (
    (kind = 'adjustment' AND amount <> 0) OR (kind <> 'adjustment' AND amount > 0)),
  -- Tiền mặt không qua cổng ⇒ không có mã cổng; ví/thẻ luôn có.
  CONSTRAINT ck_payments_ref_by_method CHECK ((method = 'cash') = (provider_ref IS NULL))
);

-- Hàm băm sinh dữ liệu: nhân Knuth (2654435761) trên 31 bit thấp, lấy 16 bit
-- cao của tích 32 bit ⇒ số nguyên 0..65535. Mọi phép tính nằm trong bigint
-- (tích lớn nhất ≈ 5,7·10^18 < 9,2·10^18), không tràn.
CREATE FUNCTION p02_hash(n bigint, salt integer) RETURNS integer
LANGUAGE sql IMMUTABLE STRICT
AS $$
  SELECT ((((n * 1000003 + salt * 7919) % 2147483648) * 2654435761) % 4294967296 / 65536)::integer
$$;

INSERT INTO payments (trip_id, kind, method, status, amount, provider_ref, paid_at)

       paid_at
FROM (

  SELECT code, 2, trip_id, 'charge',
         CASE WHEN tra_mot_phan_hop_le THEN 'wallet' ELSE method END, 'succeeded',
         CASE WHEN tra_mot_phan_hop_le THEN fare - phan_the ELSE fare END,
         CASE WHEN lan_dau_loi AND method <> 'cash' THEN '2' ELSE '1' END,
         completed_at + CASE WHEN lan_dau_loi AND method <> 'cash' THEN interval '1 minute' ELSE interval '0' END
  FROM (SELECT *, method = 'wallet' AND tra_mot_phan AND NOT lan_dau_loi AND fare > 40000 AS tra_mot_phan_hop_le

  WHERE status = 'completed' AND fare > 0

--    gắn đúng nghĩa của nó thay vì để ô rỗng.

SELECT t.code, t.kind, t.status,
       coalesce(m.name,
                CASE WHEN t.kind = 'food' THEN 'LỖI: đơn đồ ăn thiếu quán'

-- 7. Khoảng thời gian (interval): hiệu hai timestamptz. Thời gian từ lúc đặt

-- 1. Nạp lô 1 bằng UPSERT: INSERT … ON CONFLICT (cột khoá) DO UPDATE.
--    EXCLUDED là dòng ĐỊNH chèn (bị xung đột); bảng đích gọi qua bí danh m.

INSERT INTO menu_items AS m (merchant_id, sku, name, price, is_available, source_updated_at, updated_by_batch)

FROM menu_item_imports
WHERE batch_id = 1
ORDER BY line_no
ON CONFLICT (merchant_id, sku) DO UPDATE
SET name = EXCLUDED.name, price = EXCLUDED.price, is_available = EXCLUDED.is_available,

-- 6. CHỨNG MINH điều kiện trang kế tiếp: đi hết mọi trang bằng recursive CTE
--    (PG-019 giải thích cú pháp) — mỗi bước lấy 5 dòng sau con trỏ của bước
--    trước. Đúng khi: hợp các trang = toàn bộ lịch sử, KHÔNG dòng nào lặp,

WITH RECURSIVE trang (so_trang, ma, con_tro_at, con_tro_code) AS (
  SELECT 1, p.ma, p.ats[cardinality(p.ats)], p.ma[cardinality(p.ma)]
  FROM (SELECT array_agg(code ORDER BY requested_at DESC, code DESC) AS ma,
               array_agg(requested_at ORDER BY requested_at DESC, code DESC) AS ats
        FROM (SELECT code, requested_at FROM trips
              WHERE customer_id = 'cb000000-0000-4000-8000-000000000001'
              ORDER BY requested_at DESC, code DESC LIMIT 5) AS dau) AS p
  UNION ALL
  SELECT tr.so_trang + 1, nx.ma, nx.ats[cardinality(nx.ats)], nx.ma[cardinality(nx.ma)]
  FROM trang AS tr
  CROSS JOIN LATERAL (
    SELECT array_agg(code ORDER BY requested_at DESC, code DESC) AS ma,
           array_agg(requested_at ORDER BY requested_at DESC, code DESC) AS ats
    FROM (SELECT code, requested_at FROM trips
          WHERE customer_id = 'cb000000-0000-4000-8000-000000000001'
            AND (requested_at, code) < (tr.con_tro_at, tr.con_tro_code)
          ORDER BY requested_at DESC, code DESC LIMIT 5) AS sau
  ) AS nx
  WHERE nx.ma IS NOT NULL          -- hết dòng ⇒ array_agg trả NULL ⇒ dừng
)

-- 0. Sổ đối soát theo chuyến: một VIEW tạm (sống trong phiên) để các bước sau
--    dùng chung MỘT định nghĩa — không chép lại công thức ở từng truy vấn.
CREATE TEMP VIEW so_doi_soat AS
WITH pham_vi AS (
  SELECT t.id, t.code, t.kind, t.city_id, t.status, t.fare,
         (t.requested_at AT TIME ZONE 'Asia/Ho_Chi_Minh')::date AS ngay,
         CASE WHEN t.status = 'completed' THEN t.fare ELSE 0 END AS phai_thu
  FROM trips AS t
  WHERE t.status IN ('completed', 'cancelled')
    AND t.requested_at >= '2026-10-01 00:00+07' AND t.requested_at < '2026-10-31 00:00+07'
),
gd AS (
  SELECT p.trip_id,
         count(*) FILTER (WHERE kind = 'charge' AND status = 'succeeded')                 AS so_thu_ok,
         count(*) FILTER (WHERE kind = 'charge' AND status = 'succeeded')
           - count(DISTINCT amount) FILTER (WHERE kind = 'charge' AND status = 'succeeded') AS so_thu_lap_tien,
         coalesce(sum(amount) FILTER (WHERE kind = 'charge'     AND status = 'succeeded'), 0) AS tong_thu,
         coalesce(sum(amount) FILTER (WHERE kind = 'refund'     AND status = 'succeeded'), 0) AS tong_hoan,
         coalesce(sum(amount) FILTER (WHERE kind = 'adjustment' AND status = 'succeeded'), 0) AS tong_dieu_chinh,
         count(*) FILTER (WHERE status <> 'succeeded')                                    AS so_gd_khong_thanh_cong,
         string_agg(kind || '/' || method || '/' || status || '/' || amount, ', ' ORDER BY paid_at, id) AS chi_tiet
  FROM payments AS p
  GROUP BY p.trip_id
)
SELECT pv.*,
       coalesce(g.so_thu_ok, 0)                                        AS so_thu_ok,
       coalesce(g.tong_thu, 0)                                         AS tong_thu,
       coalesce(g.tong_hoan, 0)                                        AS tong_hoan,
       coalesce(g.tong_dieu_chinh, 0)                                  AS tong_dieu_chinh,
       coalesce(g.tong_thu - g.tong_hoan + g.tong_dieu_chinh, 0)       AS thuc_thu,
       coalesce(g.tong_thu - g.tong_hoan + g.tong_dieu_chinh, 0) - pv.phai_thu AS chenh,
       coalesce(g.so_gd_khong_thanh_cong, 0)                           AS so_gd_khong_thanh_cong,
       coalesce(g.chi_tiet, '(không có giao dịch)')                    AS chi_tiet,
       CASE
         WHEN coalesce(g.tong_thu - g.tong_hoan + g.tong_dieu_chinh, 0) = pv.phai_thu THEN NULL
         WHEN coalesce(g.so_thu_ok, 0) = 0 AND g.tong_hoan > 0          THEN '1 HOÀN KHÔNG CÓ THU'
         WHEN pv.status = 'cancelled'                                   THEN '2 THU CHUYẾN HUỶ'
         WHEN pv.phai_thu > 0 AND coalesce(g.so_thu_ok, 0) = 0          THEN '3 THIẾU THANH TOÁN'
         WHEN g.so_thu_lap_tien > 0                                     THEN '4 THU TRÙNG'
         ELSE                                                                '5 LỆCH SỐ TIỀN'
       END AS loai_lech
FROM pham_vi AS pv
LEFT JOIN gd AS g ON g.trip_id = pv.id;

-- 3. ĐỐI SOÁT SỐ DÒNG: mọi chuyến trong phạm vi rơi vào ĐÚNG MỘT nhóm, các
--    nhóm cộng lại = 3.168.
-- expect-line: KHỚP — hoàn thành, có thu                 | 2671 | 181941000
-- expect-line: KHỚP — chuyến huỷ, không giao dịch        |  447 |         0
-- expect-line: KHỚP — hoàn thành cước 0 đ, không cần thu |   31 |         0
-- expect-line: KHỚP — chuyến huỷ, đã hoàn đủ             |   12 |   1008000
-- expect-line: LỆCH                                      |    7 |    691000
-- expect-line: TỔNG                                      | 3168 | 183640000
SELECT coalesce(nhom, 'TỔNG') AS nhom, count(*) AS so_chuyen, sum(tong_thu) AS tong_tien_charge
FROM (
  SELECT tong_thu,
         CASE
           WHEN loai_lech IS NOT NULL                       THEN 'LỆCH'
           WHEN status = 'completed' AND phai_thu = 0       THEN 'KHỚP — hoàn thành cước 0 đ, không cần thu'
           WHEN status = 'completed'                        THEN 'KHỚP — hoàn thành, có thu'
           WHEN tong_hoan > 0                               THEN 'KHỚP — chuyến huỷ, đã hoàn đủ'
           ELSE                                                  'KHỚP — chuyến huỷ, không giao dịch'
         END AS nhom
  FROM so_doi_soat
) AS x
GROUP BY ROLLUP (nhom)
ORDER BY grouping(nhom), count(*) DESC;

-- adjustment) báo nhầm hàng trăm chuyến — đối soát sai còn tệ hơn không đối soát.
-- expect-line: 115 | 7
SELECT count(*) FILTER (WHERE ngay_tho <> phai_thu) AS quy_tac_ngay_tho,
       count(*) FILTER (WHERE loai_lech IS NOT NULL) AS quy_tac_dung

         coalesce((SELECT sum(p.amount) FROM payments AS p

-- 5. CHÊNH LỆCH THỜI ĐIỂM — không phải lỗi: gom tiền theo NGÀY GIAO DỊCH thì
--    tiền của chuyến đặt cuối ngày rơi sang ngày hôm sau. Theo ngày đặt chuyến
--    và theo ngày giao dịch (paid_at, giờ VN), hai ngày quanh 15/10:
-- expect-line: 2026-10-15 | 5862000 | 5823000
-- expect-line: 2026-10-16 | 5244000 | 5228000
SELECT ngay,
       (SELECT sum(s.thuc_thu) FROM so_doi_soat AS s WHERE s.ngay = d.ngay) AS theo_ngay_dat_chuyen,
       (SELECT sum(CASE p.kind WHEN 'refund' THEN -p.amount ELSE p.amount END)
          FROM payments AS p
         WHERE p.status = 'succeeded'
           AND (p.paid_at AT TIME ZONE 'Asia/Ho_Chi_Minh')::date = d.ngay)   AS theo_ngay_giao_dich
FROM (VALUES (DATE '2026-10-15'), (DATE '2026-10-16')) AS d (ngay)
ORDER BY ngay;

SELECT (SELECT sum(thuc_thu) FROM so_doi_soat) AS theo_ngay_dat_chuyen,

         WHERE status = 'succeeded'
           AND paid_at >= '2026-10-01 00:00+07' AND paid_at < '2026-10-31 00:00+07') AS theo_ngay_giao_dich_thang_10,
       (SELECT sum(CASE kind WHEN 'refund' THEN -amount ELSE amount END) FROM payments
         WHERE status = 'succeeded' AND paid_at >= '2026-10-31 00:00+07')            AS ghi_nhan_sau_31_10;

-- 6. VIỆC CẦN THEO DÕI (không đổi tiền, nhưng phải có người xử lý): giao dịch
--    còn `pending` sau khi chuyến đã hoàn thành hơn 1 giờ (tính tới lúc chụp
--    dữ liệu 2026-10-31 01:00).
-- expect-line: 103162 | TH-103162-P | 57000 | 2026-10-30 22:17:37+07
SELECT t.code, p.provider_ref, p.amount, p.paid_at
FROM payments AS p
JOIN trips AS t ON t.id = p.trip_id
WHERE p.status = 'pending'
  AND t.completed_at < TIMESTAMPTZ '2026-10-31 01:00+07' - interval '1 hour'
ORDER BY t.code;
